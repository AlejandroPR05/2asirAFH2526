Guía Práctica: Pre-entrenamiento de un "Nano-Modelo" de IA

Este manual está diseñado para guiarte paso a paso en el proceso de crear y pre-entrenar una Inteligencia Artificial pequeña desde cero en Windows 11. [cite_start]El objetivo no es crear un competidor de GPT-4, sino entender conceptualmente cómo aprenden estos modelos[cite: 69].

[cite_start]Concepto Clave: El pre-entrenamiento es la fase donde el modelo adquiere conocimiento general prediciendo la siguiente palabra (o carácter) en un texto masivo[cite: 3, 10]. [cite_start]En este ejercicio, nuestro "internet" será únicamente el libro Don Quijote de la Mancha[cite: 72].

1. Conceptos Fundamentales

Antes de empezar, entiende las tres piezas que vamos a construir:

  • El Dataset: El texto crudo. [cite_start]El modelo aprenderá el estilo del castellano del siglo XVII[cite: 74].
  • El Tokenizador: El traductor que convierte texto en números. [cite_start]Las IAs no leen letras, procesan secuencias numéricas[cite: 21, 22].
  • El Transformer: El cerebro de la operación. [cite_start]Una arquitectura de red neuronal que usa el mecanismo de "atención" para entender el contexto y generar texto nuevo[cite: 38, 42].

import torch
import torch.nn as nn
from torch.nn import functional as F

# --- Hiperparámetros ---
[cite_start]batch_size = 32        # Cuántas secuencias procesamos en paralelo [cite: 117]
[cite_start]block_size = 128       # Longitud máxima del contexto [cite: 118]
[cite_start]max_iters = 5000       # Número de iteraciones de entrenamiento [cite: 119]
[cite_start]eval_interval = 500    # Cada cuánto evaluamos el loss [cite: 122]
[cite_start]learning_rate = 3e-4   # Tasa de aprendizaje [cite: 123]
[cite_start]device = 'cuda' if torch.cuda.is_available() else 'cpu' # Usa GPU si tienes NVIDIA, sino CPU [cite: 124]
[cite_start]n_embd = 256           # Dimensión de los embeddings [cite: 126]
[cite_start]n_head = 4             # Número de cabezas de atención [cite: 127]
[cite_start]n_layer = 4            # Número de bloques Transformer [cite: 129]
[cite_start]dropout = 0.2          # Para evitar memorización exacta [cite: 130]

print(f"Usando dispositivo: {device}")

# --- 1. Carga y Tokenización ---
torch.manual_seed(1337)

try:
    with open('quijote.txt', 'r', encoding='utf-8') as f:
        text = f.read()
except FileNotFoundError:
    print("Error: No se encuentra 'quijote.txt'. Asegúrate de que esté en la misma carpeta.")
    exit()

# [cite_start]Crear vocabulario (caracteres únicos) [cite: 90, 91]
chars = sorted(list(set(text)))
vocab_size = len(chars)
print(f"Tamaño del vocabulario: {vocab_size} caracteres")

# [cite_start]Mapeos de caracteres a enteros y viceversa [cite: 101, 102]
stoi = {ch:i for i,ch in enumerate(chars)}
itos = {i:ch for i,ch in enumerate(chars)}
encode = lambda s: [stoi[c] for c in s] # string -> lista de enteros
decode = lambda l: ''.join([itos[i] for i in l]) # lista de enteros -> string

# [cite_start]Preparar datos para PyTorch [cite: 138]
data = torch.tensor(encode(text), dtype=torch.long)
n = int(0.9*len(data)) # 90% para entrenar, 10% para validar
train_data = data[:n]
val_data = data[n:]

# [cite_start]--- Función para obtener lotes (batches) --- [cite: 142]
def get_batch(split):
    data = train_data if split == 'train' else val_data
    ix = torch.randint(len(data) - block_size, (batch_size,))
    x = torch.stack([data[i:i+block_size] for i in ix])
    y = torch.stack([data[i+1:i+block_size+1] for i in ix])
    x, y = x.to(device), y.to(device)
    return x, y

# [cite_start]--- Estimación de pérdida (Loss) --- [cite: 150]
@torch.no_grad()
def estimate_loss():
    out = {}
    model.eval()
    for split in ['train', 'val']:
        losses = torch.zeros(200)
        for k in range(200):
            X, Y = get_batch(split)
            logits, loss = model(X, Y)
            losses[k] = loss.item()
        out[split] = losses.mean()
    model.train()
    return out

# --- Arquitectura del Modelo ---

class Head(nn.Module):
    """ Una cabeza de auto-atención """
    def __init__(self, head_size):
        super().__init__()
        self.key = nn.Linear(n_embd, head_size, bias=False)
        self.query = nn.Linear(n_embd, head_size, bias=False)
        self.value = nn.Linear(n_embd, head_size, bias=False)
        self.register_buffer('tril', torch.tril(torch.ones(block_size, block_size)))
        self.dropout = nn.Dropout(dropout)

    def forward(self, x):
        B,T,C = x.shape
        k = self.key(x)   # (B,T,head_size)
        q = self.query(x) # (B,T,head_size)
        # Calcular scores de atención ("afinidad")
        wei = q @ k.transpose(-2,-1) * C**-0.5 # (B, T, T)
        wei = wei.masked_fill(self.tril[:T, :T] == 0, float('-inf'))
        wei = F.softmax(wei, dim=-1)
        wei = self.dropout(wei)
        # Agregar los valores
        v = self.value(x)
        out = wei @ v
        return out

class MultiHeadAttention(nn.Module):
    """ Múltiples cabezas de atención en paralelo """
    def __init__(self, num_heads, head_size):
        super().__init__()
        self.heads = nn.ModuleList([Head(head_size) for _ in range(num_heads)])
        self.proj = nn.Linear(n_embd, n_embd)
        self.dropout = nn.Dropout(dropout)

    def forward(self, x):
        out = torch.cat([h(x) for h in self.heads], dim=-1)
        out = self.dropout(self.proj(out))
        return out

class FeedForward(nn.Module):
    """ Una capa simple lineal seguida de no-linealidad """
    def __init__(self, n_embd):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(n_embd, 4 * n_embd),
            nn.ReLU(),
            nn.Linear(4 * n_embd, n_embd),
            nn.Dropout(dropout),
        )

    def forward(self, x):
        return self.net(x)

class Block(nn.Module):
    """ Bloque Transformer: Comunicación seguida de Computación """
    def __init__(self, n_embd, n_head):
        super().__init__()
        head_size = n_embd // n_head
        self.sa = MultiHeadAttention(n_head, head_size)
        self.ffwd = FeedForward(n_embd)
        self.ln1 = nn.LayerNorm(n_embd)
        self.ln2 = nn.LayerNorm(n_embd)

    def forward(self, x):
        x = x + self.sa(self.ln1(x))
        x = x + self.ffwd(self.ln2(x))
        return x

class LanguageModel(nn.Module):
    def __init__(self):
        super().__init__()
        self.token_embedding_table = nn.Embedding(vocab_size, n_embd)
        self.position_embedding_table = nn.Embedding(block_size, n_embd)
        self.blocks = nn.Sequential(*[Block(n_embd, n_head=n_head) for _ in range(n_layer)])
        self.ln_f = nn.LayerNorm(n_embd) # Layer norm final
        self.lm_head = nn.Linear(n_embd, vocab_size)

    def forward(self, idx, targets=None):
        B, T = idx.shape
        tok_emb = self.token_embedding_table(idx) # (B,T,C)
        pos_emb = self.position_embedding_table(torch.arange(T, device=device)) # (T,C)
        x = tok_emb + pos_emb
        x = self.blocks(x)
        x = self.ln_f(x)
        logits = self.lm_head(x) # (B,T,vocab_size)

        if targets is None:
            loss = None
        else:
            B, T, C = logits.shape
            logits = logits.view(B*T, C)
            targets = targets.view(B*T)
            loss = F.cross_entropy(logits, targets)

        return logits, loss

    def generate(self, idx, max_new_tokens):
        # idx es un array (B, T) de índices en el contexto actual
        for _ in range(max_new_tokens):
            idx_cond = idx[:, -block_size:] # Recortar al contexto máximo
            logits, loss = self(idx_cond)
            logits = logits[:, -1, :] # Enfocarse solo en el último paso de tiempo
            probs = F.softmax(logits, dim=-1) # (B, C)
            idx_next = torch.multinomial(probs, num_samples=1) # (B, 1)
            idx = torch.cat((idx, idx_next), dim=1) # (B, T+1)
        return idx

# [cite_start]--- Bucle de Entrenamiento --- [cite: 239]
model = LanguageModel()
m = model.to(device)
optimizer = torch.optim.AdamW(model.parameters(), lr=learning_rate)

print(f"Iniciando entrenamiento por {max_iters} iteraciones...")

for iter in range(max_iters):
    # Cada cierto tiempo imprimimos el error
    if iter % eval_interval == 0:
        losses = estimate_loss()
        print(f"Iteración {iter}: train loss {losses['train']:.4f}, val loss {losses['val']:.4f}")

    # Obtener lote y entrenar
    xb, yb = get_batch('train')
    logits, loss = model(xb, yb)
    optimizer.zero_grad(set_to_none=True)
    loss.backward()
    optimizer.step()

# [cite_start]--- Generación Final --- [cite: 255]
print("\n--- ¡Entrenamiento completado! Generando texto estilo Cervantes... ---\n")
context = torch.zeros((1, 1), dtype=torch.long, device=device)
generated_text = decode(m.generate(context, max_new_tokens=500)[0].tolist())
print(generated_text)

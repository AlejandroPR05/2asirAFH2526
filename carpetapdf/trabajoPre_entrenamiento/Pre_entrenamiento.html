Guía Práctica: Pre-entrenamiento de un "Nano-Modelo" de IA

Este manual está diseñado para guiarte paso a paso en el proceso de crear y pre-entrenar una Inteligencia Artificial pequeña desde cero en Windows 11. El objetivo es entender conceptualmente cómo aprenden estos modelos utilizando el texto de Don Quijote de la Mancha.

Concepto Clave: El pre-entrenamiento es la fase donde el modelo adquiere conocimiento general intentando predecir incansablemente cuál es la siguiente palabra (o carácter) en un texto masivo.

1. Conceptos Fundamentales y Flujo de Trabajo

Antes de empezar, visualiza las tres piezas clave que vamos a construir y cómo interactúan en tu PC:

  • El Dataset: Es la "comida" de la IA. El texto crudo (`quijote.txt`). El modelo aprenderá el estilo y vocabulario del castellano del siglo XVII.
  • El Tokenizador: Es el "traductor". Convierte el texto humano en secuencias de números que la máquina puede entender, y viceversa.
  • El Transformer (Modelo): Es el "cerebro". Una red neuronal que procesa esos números, encuentra patrones complejos y aprende a predecir qué viene después.

import torch
import torch.nn as nn
from torch.nn import functional as F

# --- Hiperparámetros ---
# Ajustan el comportamiento y tamaño del modelo
batch_size = 32        # Cuántas secuencias procesamos en paralelo
block_size = 128       # Longitud máxima del contexto (memoria a corto plazo)
max_iters = 5000       # Número total de pasos de entrenamiento
eval_interval = 500    # Cada cuántos pasos medimos el error
learning_rate = 3e-4   # Velocidad a la que aprende el modelo
device = 'cuda' if torch.cuda.is_available() else 'cpu' # Usa GPU Nvidia si existe, si no CPU
n_embd = 256           # Dimensión de los vectores de embedding
n_head = 4             # Número de cabezas de atención paralelas
n_layer = 4            # Número de bloques Transformer apilados
dropout = 0.2          # Técnica para evitar que el modelo solo memorice

print(f"--> Usando dispositivo para entrenamiento: {device}")

# --- 1. Carga y Tokenización ---
torch.manual_seed(1337) # Para resultados reproducibles

try:
    with open('quijote.txt', 'r', encoding='utf-8') as f:
        text = f.read()
except FileNotFoundError:
    print("\n[ERROR CRÍTICO]: No se encuentra 'quijote.txt'.")
    print("Asegúrate de que el archivo esté en la misma carpeta que este script.")
    exit()

# Crear vocabulario (todos los caracteres únicos que aparecen)
chars = sorted(list(set(text)))
vocab_size = len(chars)
print(f"--> Tamaño del vocabulario: {vocab_size} caracteres únicos detectados.")

# Mapeos: Diccionarios para traducir de caracteres a enteros y viceversa
stoi = {ch:i for i,ch in enumerate(chars)}
itos = {i:ch for i,ch in enumerate(chars)}
encode = lambda s: [stoi[c] for c in s] # Función: texto -> lista de enteros
decode = lambda l: ''.join([itos[i] for i in l]) # Función: lista de enteros -> texto

# Preparar datos para PyTorch
print("--> Tokenizando el texto y moviéndolo a tensores...")
data = torch.tensor(encode(text), dtype=torch.long)
n = int(0.9*len(data)) # Usamos el 90% para entrenar y el 10% para validar
train_data = data[:n]
val_data = data[n:]

# --- Función para obtener lotes (batches) aleatorios ---
def get_batch(split):
    data = train_data if split == 'train' else val_data
    # Elegimos puntos de inicio aleatorios en el texto
    ix = torch.randint(len(data) - block_size, (batch_size,))
    # Creamos los bloques de contexto (X) y los objetivos a predecir (Y)
    x = torch.stack([data[i:i+block_size] for i in ix])
    y = torch.stack([data[i+1:i+block_size+1] for i in ix])
    x, y = x.to(device), y.to(device) # Movemos los datos a la GPU/CPU
    return x, y

# --- Estimación de pérdida (Loss) ---
# Esta función mide qué tan mal lo está haciendo el modelo sin entrenarlo
@torch.no_grad()
def estimate_loss():
    out = {}
    model.eval() # Ponemos el modelo en modo evaluación
    for split in ['train', 'val']:
        losses = torch.zeros(200)
        for k in range(200): # Promediamos 200 lotes para ser precisos
            X, Y = get_batch(split)
            logits, loss = model(X, Y)
            losses[k] = loss.item()
        out[split] = losses.mean()
    model.train() # Devolvemos el modelo a modo entrenamiento
    return out

# --- Arquitectura del Modelo Transformer ---

class Head(nn.Module):
    """ Una sola cabeza de auto-atención """
    def __init__(self, head_size):
        super().__init__()
        self.key = nn.Linear(n_embd, head_size, bias=False)
        self.query = nn.Linear(n_embd, head_size, bias=False)
        self.value = nn.Linear(n_embd, head_size, bias=False)
        self.register_buffer('tril', torch.tril(torch.ones(block_size, block_size)))
        self.dropout = nn.Dropout(dropout)

    def forward(self, x):
        B,T,C = x.shape
        k = self.key(x)   # (B,T,head_size)
        q = self.query(x) # (B,T,head_size)
        # Calcular scores de atención ("afinidad" entre tokens)
        wei = q @ k.transpose(-2,-1) * C**-0.5 # (B, T, T)
        # Enmascarar el futuro: no dejar que mire adelante
        wei = wei.masked_fill(self.tril[:T, :T] == 0, float('-inf'))
        wei = F.softmax(wei, dim=-1)
        wei = self.dropout(wei)
        # Agregar los valores ponderados
        v = self.value(x)
        out = wei @ v
        return out

class MultiHeadAttention(nn.Module):
    """ Múltiples cabezas de atención trabajando en paralelo """
    def __init__(self, num_heads, head_size):
        super().__init__()
        self.heads = nn.ModuleList([Head(head_size) for _ in range(num_heads)])
        self.proj = nn.Linear(n_embd, n_embd)
        self.dropout = nn.Dropout(dropout)

    def forward(self, x):
        # Concatenamos los resultados de todas las cabezas
        out = torch.cat([h(x) for h in self.heads], dim=-1)
        out = self.dropout(self.proj(out))
        return out

class FeedForward(nn.Module):
    """ Una red neuronal simple (MLP) dentro del bloque """
    def __init__(self, n_embd):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(n_embd, 4 * n_embd),
            nn.ReLU(), # Función de activación no lineal
            nn.Linear(4 * n_embd, n_embd),
            nn.Dropout(dropout),
        )

    def forward(self, x):
        return self.net(x)

class Block(nn.Module):
    """ Bloque Transformer estándar: Comunicación seguida de Computación """
    def __init__(self, n_embd, n_head):
        super().__init__()
        head_size = n_embd // n_head
        self.sa = MultiHeadAttention(n_head, head_size)
        self.ffwd = FeedForward(n_embd)
        self.ln1 = nn.LayerNorm(n_embd) # Normalización por capas
        self.ln2 = nn.LayerNorm(n_embd)

    def forward(self, x):
        # Conexiones residuales (x + ...)
        x = x + self.sa(self.ln1(x))
        x = x + self.ffwd(self.ln2(x))
        return x

class LanguageModel(nn.Module):
    """ El modelo completo ensamblado """
    def __init__(self):
        super().__init__()
        # Tabla para convertir tokens en vectores
        self.token_embedding_table = nn.Embedding(vocab_size, n_embd)
        # Tabla para codificar la posición de cada token
        self.position_embedding_table = nn.Embedding(block_size, n_embd)
        # La secuencia de bloques Transformer
        self.blocks = nn.Sequential(*[Block(n_embd, n_head=n_head) for _ in range(n_layer)])
        self.ln_f = nn.LayerNorm(n_embd) # Layer norm final
        # La capa final que convierte de nuevo a probabilidades del vocabulario
        self.lm_head = nn.Linear(n_embd, vocab_size)

    def forward(self, idx, targets=None):
        B, T = idx.shape
        tok_emb = self.token_embedding_table(idx) # (B,T,C)
        pos_emb = self.position_embedding_table(torch.arange(T, device=device)) # (T,C)
        x = tok_emb + pos_emb # Sumamos embeddings de token y posición
        x = self.blocks(x)    # Pasamos por los bloques Transformer
        x = self.ln_f(x)
        logits = self.lm_head(x) # (B,T,vocab_size)

        if targets is None:
            loss = None
        else:
            # Calculamos el error (cross_entropy) si tenemos objetivos
            B, T, C = logits.shape
            logits = logits.view(B*T, C)
            targets = targets.view(B*T)
            loss = F.cross_entropy(logits, targets)

        return logits, loss

    def generate(self, idx, max_new_tokens):
        """ Función para generar texto nuevo una vez entrenado """
        # idx es el contexto actual (B, T)
        for _ in range(max_new_tokens):
            # Recortamos el contexto si excede el tamaño del bloque
            idx_cond = idx[:, -block_size:]
            # Obtenemos las predicciones
            logits, loss = self(idx_cond)
            # Nos enfocamos solo en el último paso de tiempo (la predicción actual)
            logits = logits[:, -1, :]
            # Convertimos logits a probabilidades
            probs = F.softmax(logits, dim=-1)
            # Muestreamos la siguiente letra basándonos en las probabilidades
            idx_next = torch.multinomial(probs, num_samples=1)
            # Agregamos la nueva letra al contexto y repetimos
            idx = torch.cat((idx, idx_next), dim=1)
        return idx

# --- Bucle Principal de Entrenamiento ---
print("\n--> Inicializando el modelo...")
model = LanguageModel()
m = model.to(device) # Mover modelo a GPU/CPU
# Optimizador AdamW: el algoritmo que ajusta los pesos de la red
optimizer = torch.optim.AdamW(model.parameters(), lr=learning_rate)

print(f"--> Iniciando entrenamiento intensivo por {max_iters} iteraciones. Por favor espera...\n")

for iter in range(max_iters):
    # Cada cierto intervalo, evaluamos y mostramos el progreso
    if iter % eval_interval == 0 or iter == max_iters - 1:
        losses = estimate_loss()
        print(f"Iteración {iter}: train loss {losses['train']:.4f}, val loss {losses['val']:.4f}")

    # Obtener un lote de datos
    xb, yb = get_batch('train')
    
    # Forward pass (calcular predicciones y error)
    logits, loss = model(xb, yb)
    
    # Backward pass (calcular gradientes y actualizar pesos)
    optimizer.zero_grad(set_to_none=True)
    loss.backward()
    optimizer.step()

# --- Generación Final ---
print("\n--------------------------------------------------")
print(" ¡Entrenamiento completado! ")
print(" El modelo ahora intentará escribir como Cervantes:")
print("--------------------------------------------------\n")

# Creamos un contexto inicial vacío (un carácter de nueva línea)
context = torch.zeros((1, 1), dtype=torch.long, device=device)
# Generamos 500 tokens nuevos y los decodificamos a texto
generated_text = decode(m.generate(context, max_new_tokens=500)[0].tolist())
print(generated_text)
